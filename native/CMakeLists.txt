cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../amazing_overlay/version.json" version_json)
string(JSON HELPER_VERSION GET "${version_json}" version)
project(AmazingLawClient VERSION ${HELPER_VERSION} LANGUAGES C CXX)
configure_file(helper_version.h.in helper_version.h @ONLY)
enable_testing()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "The installed Amazing client is x86. Configure with -A Win32.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.91.9b GIT_SHALLOW TRUE)
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG v1.3.4 GIT_SHALLOW TRUE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(imgui minhook json)
include(blackbone.cmake)
add_executable(AmazingHelperInjector WIN32 injector.cpp)
target_include_directories(AmazingHelperInjector PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(AmazingHelperInjector PRIVATE /utf-8)
target_link_libraries(AmazingHelperInjector PRIVATE blackbone_user nlohmann_json::nlohmann_json bcrypt shell32)
add_executable(overlay_test_injector WIN32 injector.cpp)
target_include_directories(overlay_test_injector PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(overlay_test_injector PRIVATE AMAZING_HELPER_TEST_HOST)
target_compile_options(overlay_test_injector PRIVATE /utf-8)
target_link_libraries(overlay_test_injector PRIVATE blackbone_user nlohmann_json::nlohmann_json bcrypt shell32)
add_library(overlay_imgui STATIC
  ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_dx9.cpp)
target_include_directories(overlay_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(overlay_imgui PUBLIC d3d9 dwmapi)
add_library(AmazingLawClient SHARED client.cpp rpc.cpp document_renderer.cpp settings_panel.cpp)
target_include_directories(AmazingLawClient PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(AmazingLawClient PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX DIRECTINPUT_VERSION=0x0800)
target_link_libraries(AmazingLawClient PRIVATE overlay_imgui minhook nlohmann_json::nlohmann_json d3d9 dinput8 dxguid user32 shell32)
if(MSVC)
  target_compile_options(AmazingLawClient PRIVATE /utf-8 /W4 /permissive-)
endif()
add_custom_command(TARGET AmazingLawClient POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:AmazingLawClient> $<TARGET_FILE_DIR:AmazingLawClient>/AmazingLawClient.asi)
add_executable(overlay_native_smoke WIN32 smoke.cpp)
add_executable(overlay_input_test input_policy_test.cpp)
add_test(NAME input_policy COMMAND overlay_input_test)
add_executable(overlay_renderer_test renderer_test.cpp document_renderer.cpp)
target_link_libraries(overlay_renderer_test PRIVATE overlay_imgui nlohmann_json::nlohmann_json)
target_compile_definitions(overlay_renderer_test PRIVATE NOMINMAX)
add_test(NAME document_renderer COMMAND overlay_renderer_test)
add_executable(overlay_settings_test settings_test.cpp settings_panel.cpp)
target_link_libraries(overlay_settings_test PRIVATE overlay_imgui nlohmann_json::nlohmann_json)
target_compile_definitions(overlay_settings_test PRIVATE NOMINMAX)
add_test(NAME native_settings COMMAND overlay_settings_test)
add_executable(overlay_notebook_test notebook_test.cpp)
target_link_libraries(overlay_notebook_test PRIVATE overlay_imgui)
target_compile_definitions(overlay_notebook_test PRIVATE NOMINMAX)
if(MSVC)
  target_compile_options(overlay_notebook_test PRIVATE /utf-8 /UNDEBUG)
endif()
add_test(NAME notebook_editor COMMAND overlay_notebook_test)
add_executable(overlay_desktop_ui_test desktop_ui_test.cpp)
target_link_libraries(overlay_desktop_ui_test PRIVATE overlay_imgui)
if(MSVC)
  target_compile_options(overlay_desktop_ui_test PRIVATE /utf-8)
endif()
add_test(NAME native_components COMMAND overlay_desktop_ui_test)
add_executable(overlay_chat_test chat_test.cpp)
target_link_libraries(overlay_chat_test PRIVATE overlay_imgui nlohmann_json::nlohmann_json)
target_compile_definitions(overlay_chat_test PRIVATE NOMINMAX)
if(MSVC)
  target_compile_options(overlay_chat_test PRIVATE /utf-8)
endif()
add_test(NAME native_chat COMMAND overlay_chat_test)
target_compile_definitions(overlay_native_smoke PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_link_libraries(overlay_native_smoke PRIVATE d3d9 user32)
if(MSVC)
  target_compile_options(overlay_input_test PRIVATE /utf-8)
  target_compile_options(overlay_renderer_test PRIVATE /utf-8)
  target_compile_options(overlay_settings_test PRIVATE /utf-8)
  target_compile_options(overlay_native_smoke PRIVATE /utf-8)
endif()
